cmake_minimum_required(VERSION 3.28)
project(course12_redis LANGUAGES C CXX)
set(COURSE_CXX_STANDARD "20" CACHE STRING "C++ standard for the course: 20 or 23")
set_property(CACHE COURSE_CXX_STANDARD PROPERTY STRINGS 20 23)
if(NOT COURSE_CXX_STANDARD MATCHES "^(20|23)$")
  message(FATAL_ERROR "COURSE_CXX_STANDARD must be 20 or 23")
endif()
set(CMAKE_CXX_STANDARD "${COURSE_CXX_STANDARD}")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(cache_model cache_model.cpp)
add_test(NAME cache_model COMMAND cache_model)
option(COURSE12_WITH_HIREDIS "Download hiredis 1.3.0 and build the real Redis client" OFF)
if(COURSE12_WITH_HIREDIS)
  include(FetchContent)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(ENABLE_SSL OFF CACHE BOOL "" FORCE)
  set(DISABLE_TESTS ON CACHE BOOL "" FORCE)
  FetchContent_Declare(hiredis GIT_REPOSITORY https://github.com/redis/hiredis.git
                      GIT_TAG v1.3.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(hiredis)
  add_executable(cache_client cache_client.cpp)
  target_link_libraries(cache_client PRIVATE hiredis)
endif()
