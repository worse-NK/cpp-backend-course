cmake_minimum_required(VERSION 3.22)
project(course_bounded_asio LANGUAGES CXX)
set(COURSE_CXX_STANDARD "20" CACHE STRING "Course edition: 20 or 23")
if(NOT COURSE_CXX_STANDARD MATCHES "^(20|23)$")
  message(FATAL_ERROR "COURSE_CXX_STANDARD must be 20 or 23")
endif()
set(Boost_NO_BOOST_CMAKE ON)
find_package(Boost 1.85 EXACT REQUIRED)
find_package(Threads REQUIRED)
add_executable(bounded_server server.cpp)
set_target_properties(bounded_server PROPERTIES CXX_STANDARD ${COURSE_CXX_STANDARD}
  CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_link_libraries(bounded_server PRIVATE Boost::headers Threads::Threads)
target_compile_options(bounded_server PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME loopback_protocol COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/verify.py $<TARGET_FILE:bounded_server>)
set_tests_properties(loopback_protocol PROPERTIES TIMEOUT 25)
