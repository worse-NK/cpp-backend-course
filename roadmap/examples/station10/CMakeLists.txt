cmake_minimum_required(VERSION 3.20)
project(station10_gate LANGUAGES CXX)
set(COURSE_CXX_STANDARD "20" CACHE STRING "C++ standard for the course: 20 or 23")
set_property(CACHE COURSE_CXX_STANDARD PROPERTY STRINGS 20 23)
if(NOT COURSE_CXX_STANDARD MATCHES "^(20|23)$")
  message(FATAL_ERROR "COURSE_CXX_STANDARD must be 20 or 23")
endif()
set(CMAKE_CXX_STANDARD "${COURSE_CXX_STANDARD}")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(gate_check main.cpp)
target_compile_features(gate_check PRIVATE cxx_std_20)
set_target_properties(gate_check PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME gate_contract COMMAND gate_check)
set_tests_properties(gate_contract PROPERTIES TIMEOUT 5)
