cmake_minimum_required(VERSION 3.20)
project(course_unix_ipc LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This experiment demonstrates Linux AF_UNIX APIs")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
foreach(name socketpair_stream descriptor_transfer)
    add_executable(${name} ${name}.cpp)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
enable_testing()
add_test(NAME local_stream COMMAND socketpair_stream)
add_test(NAME local_rights COMMAND descriptor_transfer)
set_tests_properties(local_stream local_rights PROPERTIES TIMEOUT 10)
