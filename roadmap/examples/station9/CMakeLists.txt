cmake_minimum_required(VERSION 3.20)
project(course_task_edit LANGUAGES CXX)
set(COURSE_CXX_STANDARD "20" CACHE STRING "C++ standard for the course: 20 or 23")
set_property(CACHE COURSE_CXX_STANDARD PROPERTY STRINGS 20 23)
if(NOT COURSE_CXX_STANDARD MATCHES "^(20|23)$")
  message(FATAL_ERROR "COURSE_CXX_STANDARD must be 20 or 23")
endif()
set(CMAKE_CXX_STANDARD "${COURSE_CXX_STANDARD}")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PQXX REQUIRED IMPORTED_TARGET libpqxx=7.10.1)
pkg_check_modules(PQ REQUIRED IMPORTED_TARGET libpq)
add_executable(task_edit main.cpp)
target_compile_features(task_edit PRIVATE cxx_std_20)
target_link_libraries(task_edit PRIVATE PkgConfig::PQXX PkgConfig::PQ)
add_executable(reserve_slot parking.cpp)
target_compile_features(reserve_slot PRIVATE cxx_std_20)
target_link_libraries(reserve_slot PRIVATE PkgConfig::PQXX PkgConfig::PQ)
