cmake_minimum_required(VERSION 3.22)
project(course11_messaging LANGUAGES CXX)
set(COURSE_CXX_STANDARD "20" CACHE STRING "C++ standard for the course: 20 or 23")
set_property(CACHE COURSE_CXX_STANDARD PROPERTY STRINGS 20 23)
if(NOT COURSE_CXX_STANDARD MATCHES "^(20|23)$")
  message(FATAL_ERROR "COURSE_CXX_STANDARD must be 20 or 23")
endif()
set(CMAKE_CXX_STANDARD "${COURSE_CXX_STANDARD}")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(delivery_model delivery_model.cpp)
enable_testing()
add_test(NAME delivery_model COMMAND delivery_model)
option(COURSE11_BUILD_CLIENTS "Build clients for an already installed Kafka/RabbitMQ SDK" OFF)
if(COURSE11_BUILD_CLIENTS)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(RDKAFKA REQUIRED IMPORTED_TARGET rdkafka>=2.12.1)
  pkg_check_modules(RABBITMQ REQUIRED IMPORTED_TARGET librabbitmq>=0.18.1)
  add_executable(kafka_roundtrip kafka_roundtrip.cpp)
  target_link_libraries(kafka_roundtrip PRIVATE PkgConfig::RDKAFKA)
  add_executable(rabbit_roundtrip rabbit_roundtrip.cpp)
  target_link_libraries(rabbit_roundtrip PRIVATE PkgConfig::RABBITMQ)
endif()
