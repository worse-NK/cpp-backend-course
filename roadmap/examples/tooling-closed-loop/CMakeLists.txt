cmake_minimum_required(VERSION 3.28)
project(course_tooling LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "This diagnostic lab is checked for GNU C++ on Linux")
endif()
find_package(Threads REQUIRED)
include(CTest)
add_executable(report_before report_before.cpp)
add_executable(report_fixed report_fixed.cpp)
foreach(target report_before report_fixed)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_options(${target} PRIVATE -O0 -g3 -Wall -Wextra -fno-omit-frame-pointer)
  target_link_options(${target} PRIVATE -Wl,--build-id=sha1)
endforeach()
add_executable(perf_workload perf_workload.cpp)
target_compile_features(perf_workload PRIVATE cxx_std_20)
target_compile_options(perf_workload PRIVATE -O2 -g -Wall -Wextra -fno-inline -fno-omit-frame-pointer)
target_link_libraries(perf_workload PRIVATE Threads::Threads)
if(BUILD_TESTING)
  add_test(NAME empty_and_nonempty_regression COMMAND report_fixed --self-test)
  set_tests_properties(empty_and_nonempty_regression PROPERTIES PASS_REGULAR_EXPRESSION "regression: OK")
endif()
