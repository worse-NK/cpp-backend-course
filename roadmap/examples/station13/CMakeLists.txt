cmake_minimum_required(VERSION 3.24)
project(course_kubernetes_lab LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This laboratory uses Linux signals; build it on Linux or inside its container.")
endif()
set(COURSE_CXX_STANDARD "20" CACHE STRING "C++ standard for the course: 20 or 23")
set_property(CACHE COURSE_CXX_STANDARD PROPERTY STRINGS 20 23)
if(NOT COURSE_CXX_STANDARD MATCHES "^(20|23)$")
  message(FATAL_ERROR "COURSE_CXX_STANDARD must be 20 or 23")
endif()
set(CMAKE_CXX_STANDARD "${COURSE_CXX_STANDARD}")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
find_package(Threads REQUIRED)
set(HTTPLIB_USE_OPENSSL_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZLIB_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_BROTLI_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_USE_ZSTD_IF_AVAILABLE OFF CACHE BOOL "" FORCE)
set(HTTPLIB_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(httplib
  URL https://codeload.github.com/yhirose/cpp-httplib/tar.gz/refs/tags/v0.59.0
  URL_HASH SHA256=7c8cc7df044abb837d75f7c1e56333305acb40335c3788b8f5fbf5927e63e148
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(httplib)
add_executable(server app.cpp)
target_link_libraries(server PRIVATE httplib::httplib Threads::Threads)
target_compile_options(server PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(server PRIVATE
  CPPHTTPLIB_HEADER_MAX_LENGTH=4096 CPPHTTPLIB_HEADER_MAX_COUNT=32
  CPPHTTPLIB_REQUEST_URI_MAX_LENGTH=512)
add_executable(drain_model drain_model.cpp)
include(CTest)
add_test(NAME admission_contract COMMAND server --self-test)
add_test(NAME drain_trace COMMAND drain_model)
