cmake_minimum_required(VERSION 3.24)
project(course_release_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PostgreSQL REQUIRED)
option(RELEASE_USE_INSTALLED_PQXX "Use an already installed exact libpqxx 7.10.1" OFF)
if(RELEASE_USE_INSTALLED_PQXX)
  find_package(libpqxx 7.10.1 EXACT CONFIG REQUIRED)
else()
  include(FetchContent)
  set(BUILD_TEST OFF CACHE BOOL "" FORCE)
  set(SKIP_BUILD_TEST ON CACHE BOOL "" FORCE)
  set(BUILD_DOC OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF)
  FetchContent_Declare(libpqxx
    URL https://codeload.github.com/jtv/libpqxx/tar.gz/refs/tags/7.10.1
    URL_HASH SHA256=cfbbb1d93a0a3d81319ec71d9a3db80447bb033c4f6cee088554a88862fd77d7
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(libpqxx)
endif()
foreach(release_version 1 2)
  add_executable(release_v${release_version} client.cpp)
  target_compile_definitions(release_v${release_version} PRIVATE RELEASE_VERSION=${release_version})
  target_link_libraries(release_v${release_version} PRIVATE libpqxx::pqxx)
  target_compile_options(release_v${release_version} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
